const AWS = require('aws-sdk');
const ec2 = new AWS.EC2();

// EC2 instance configuration
const instanceParams = {
  ImageId: 'ami-0c55b159cbfafe1f0', // Amazon Linux 2 AMI ID
  InstanceType: 't2.micro',
  MinCount: 1,
  MaxCount: 1,
  KeyName: 'my-key-pair',
  SecurityGroupIds: ['sg-12345678'],
  TagSpecifications: [
    {
      ResourceType: 'instance',
      Tags: [
        {
          Key: 'Name',
          Value: 'MyEC2Instance'
        }
      ]
    }
  ]
};

// Function to launch EC2 instance
async function launchEC2Instance() {
  try {
    const data = await ec2.runInstances(instanceParams).promise();
    console.log('Instance launched:', data.Instances[0].InstanceId);
    return data.Instances[0].InstanceId;
  } catch (err) {
    console.error('Error launching instance:', err);
    throw err;
  }
}
